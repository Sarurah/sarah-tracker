Webhook Setup Instructions

  1. Open Google Sheets and create a new spreadsheet.
  2. Click on Extensions > Apps Script in the top menu.
  3. Delete any existing code and paste the script below into the code editor.
  4. In the top right corner of Apps Script, click Deploy > New deployment.
  5. Click the gear icon next to Select type and select Web app.
  6. Set "Execute as" to Me and "Who has access" to Anyone.
  7. Click Deploy, authorize permissions, and copy the generated Web App URL.
  8. Paste the URL into the input field above and click Save Webhook URL.
Google Apps Script (Code.gs) 
function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    if (contents.history && Array.isArray(contents.history)) {
      var sheet = ss.getSheetByName('History') || ss.insertSheet('History');
      sheet.clear();
      sheet.appendRow(['ID', 'Title', 'Category', 'Reward ($)', 'Timestamp']);
      contents.history.forEach(function(item) {
        sheet.appendRow([item.id, item.title, item.category, item.reward, item.timestamp]);
      });
    }

    if (contents.healthLogs && contents.healthLogs.water) {
      var wSheet = ss.getSheetByName('Health_Water') || ss.insertSheet('Health_Water');
      wSheet.clear();
      wSheet.appendRow(['ID', 'Amount (Glasses)', 'Timestamp']);
      contents.healthLogs.water.forEach(function(w) {
        wSheet.appendRow([w.id, w.amount || 1, w.timestamp]);
      });
    }

    if (contents.healthLogs && contents.healthLogs.poop) {
      var pSheet = ss.getSheetByName('Health_Poop') || ss.insertSheet('Health_Poop');
      pSheet.clear();
      pSheet.appendRow(['ID', 'Timestamp']);
      contents.healthLogs.poop.forEach(function(p) {
        pSheet.appendRow([p.id, p.timestamp]);
      });
    }

    return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
